% soft shadow: stacked translucent rounded rectangles
\newcommand{\softshadow}[2]{\foreach \k in {1,...,5}{\fill[ink, opacity=.035, rounded corners=\RR+\k*.3pt]
  ($(#1.south west)+(-\k*.28pt,-\k*.42pt)$) rectangle ($(#2.north east)+(\k*.28pt,\k*.1pt)$);}}
% rounded image tile; #1 extra node options, #2 name, #3 file, #4 size
\newcommand{\tile}[4][]{%
  \node[minimum size=#4, #1] (#2) {};
  \softshadow{#2}{#2}
  \begin{scope}\clip[rounded corners=\RR] (#2.south west) rectangle (#2.north east);
    \node at (#2) {\includegraphics[width=#4,height=#4]{#3}};\end{scope}
  \draw[white, line width=.55pt, rounded corners=\RR] (#2.south west) rectangle (#2.north east);}
\newcommand{\stacktile}[4][]{%
  \node[minimum size=#4, #1] (#2) {};
  \foreach \i in {3,2,1}{%
    \draw[draw=hair, line width=.35pt, fill=white, rounded corners=\RR]
      ($(#2.south west)+(\i*\S,\i*\S)$) rectangle ($(#2.north east)+(\i*\S,\i*\S)$);}
  \softshadow{#2}{#2}
  \begin{scope}\clip[rounded corners=\RR] (#2.south west) rectangle (#2.north east);
    \node at (#2) {\includegraphics[width=#4,height=#4]{#3}};\end{scope}
  \draw[white, line width=.55pt, rounded corners=\RR] (#2.south west) rectangle (#2.north east);}
